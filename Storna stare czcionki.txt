        function switchTab(tabId, btn) {
            document.querySelectorAll('.program-tab-content').forEach(content => {
                content.classList.remove('active');
            });
            document.querySelectorAll('.program-tab-btn').forEach(button => {
                button.classList.remove('active');
            });
            document.getElementById('tab-' + tabId).classList.add('active');
            btn.classList.add('active');
        }

        function openModal() {
            document.getElementById('recruitModal').style.display = 'flex';
        }
        function closeModal() {
            document.getElementById('recruitModal').style.display = 'none';
            setTimeout(() => {
                document.getElementById('formContainer').style.display = 'block';
                document.getElementById('formSuccessMessage').style.display = 'none';
                document.getElementById('recruitmentForm').reset();
                document.getElementById('submitBtn').disabled = false;
                document.getElementById('submitBtn').textContent = 'Wyślij zgłoszenie';
            }, 300);
        }
        window.onclick = function(e) {
            const modal = document.getElementById('recruitModal');
            if (e.target === modal) closeModal();
        }

        const recruitmentForm = document.getElementById('recruitmentForm');
        recruitmentForm.addEventListener('submit', function(e) {
            e.preventDefault();
            const submitBtn = document.getElementById('submitBtn');
            submitBtn.disabled = true;
            submitBtn.textContent = 'Wysyłanie...';

            const formData = new FormData(recruitmentForm);

            fetch(recruitmentForm.action, {
                method: 'POST',
                body: formData,
                headers: {
                    'Accept': 'application/json'
                }
            }).then(response => {
                if (response.ok) {
                    document.getElementById('formContainer').style.display = 'none';
                    document.getElementById('formSuccessMessage').style.display = 'block';
                } else {
                    alert('Wystąpił błąd podczas wysyłania formularza. Spróbuj ponownie lub skontaktuj się bezpośrednio z drużynowym.');
                    submitBtn.disabled = false;
                    submitBtn.textContent = 'Wyślij zgłoszenie';
                }
            }).catch(error => {
                alert('Błąd sieci. Sprawdź połączenie z internetem i spróbuj ponownie.');
                submitBtn.disabled = false;
                submitBtn.textContent = 'Wyślij zgłoszenie';
            });
        });

        const sections = document.querySelectorAll('section, header.hero');
        const navLinks = document.querySelectorAll('.nav-links a');

        window.addEventListener('scroll', () => {
            let current = '';
            const scrollPos = window.scrollY + 200;

            sections.forEach(section => {
                const sectionTop = section.offsetTop;
                const sectionHeight = section.offsetHeight;
                if (scrollPos >= sectionTop && scrollPos < sectionTop + sectionHeight) {
                    current = section.getAttribute('id');
                }
            });

            navLinks.forEach(link => {
                link.classList.remove('active-nav');
                if (link.getAttribute('href') === '#' + current) {
                    link.classList.add('active-nav');
                }
            });
        });

        const galleryItems = document.querySelectorAll('.gallery-item img');
        const lightbox = document.getElementById('lightbox');
        const lightboxImg = document.getElementById('lightbox-img');
        const lightboxClose = document.getElementById('lightbox-close');

        galleryItems.forEach(item => {
            item.addEventListener('click', () => {
                lightboxImg.setAttribute('src', item.getAttribute('src'));
                lightbox.classList.add('active');
            });
        });

        lightboxClose.addEventListener('click', () => lightbox.classList.remove('active'));
        lightbox.addEventListener('click', (e) => { if (e.target === lightbox) lightbox.classList.remove('active'); });

        let mapInitialized = false;
        const mapContainer = document.getElementById('map');

        const initMap = () => {
            if (mapInitialized) return;
            mapInitialized = true;

            var map = L.map('map', {
                dragging: !L.Browser.mobile
            });

            L.tileLayer('https://{s}.basemaps.cartocdn.com/rastertiles/voyager/{z}/{x}/{y}{r}.png', {
                attribution: '&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a> contributors &copy; <a href="https://carto.com/attributions">CARTO</a>',
                subdomains: 'abcd',
                maxZoom: 19
            }).addTo(map);

            function createColorIcon(color) {
                return L.divIcon({
                    className: 'custom-pin',
                    html: `<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 24 24" width="36" height="36" fill="${color}"><path d="M12 2C8.13 2 5 5.13 5 9c0 5.25 7 13 7 13s7-7.75 7-13c0-3.87-3.13-7-7-7z"/><circle cx="12" cy="9" r="2.5" fill="rgba(0,0,0,0.4)"/></svg>`,
                    iconSize: [36, 36],
                    iconAnchor: [18, 36],
                    popupAnchor: [0, -36]
                });
            }

            var group = L.featureGroup([marker1, marker2, marker3]).addTo(map);
            map.fitBounds(group.getBounds().pad(0.2));
        };

        const observer = new IntersectionObserver((entries, observerInstance) => {
            entries.forEach(entry => {
                if (entry.isIntersecting) {
                    initMap();
                    observerInstance.disconnect();
                }
            });
        }, { rootMargin: '200px 0px' });

        observer.observe(mapContainer);
